    async function authenticateWithPi() {

      const userDisplay = document.getElementById('user-display');

      try {

        if (!window.Pi) {

          userDisplay.innerText = "Pi SDK non-existent (Offline/Browser)";

        // Await Pi.init as a Promise

        await Pi.init({ version: "2.0", sandbox: true });

        // Authenticate using username scope

        const auth = await Pi.authenticate(['username'], onIncompletePaymentFound);

       

        // Verify access token with backend server

        const res = await fetch('/api/authenticate', {

          method: 'POST',

          headers: { 'Content-Type': 'application/json' },

          body: JSON.stringify({ accessToken: auth.accessToken })

        });

        if (!res.ok) {

          throw new Error('Backend token verification failed');

        }

        const data = await res.json();

        piUser = data.user;

        userDisplay.innerText = `Chef: ${piUser.username}`;

        loadUserData();

      } catch (err) {

        console.error("Pi Authentication failed:", err);

        userDisplay.innerText = "Authentication Failed";

        loadUserData();

      }

    }

    class SoundFx {

      playCatchGood() {

        if (this.ctx.state === 'suspended') this.ctx.resume();

        const osc = this.ctx.createOscillator();

        const gain = this.ctx.createGain();

        osc.type = 'sine';

        osc.frequency.setValueAtTime(440, this.ctx.currentTime);

        osc.frequency.exponentialRampToValueAtTime(880, this.ctx.currentTime + 0.15);

        gain.gain.setValueAtTime(0.3, this.ctx.currentTime);

        gain.gain.exponentialRampToValueAtTime(0.01, this.ctx.currentTime + 0.15);

        osc.connect(gain);

        gain.connect(this.ctx.destination);

        osc.start();

        osc.stop(this.ctx.currentTime + 0.15);

        gain.gain.setValueAtTime(0.3, this.ctx.currentTime);

        gain.gain.linearRampToValueAtTime(0.01, this.ctx.currentTime + 0.2);

        osc.connect(gain);

        gain.connect(this.ctx.destination);

        osc.stop(this.ctx.currentTime + 0.2);

        document.getElementById('hud').classList.remove('hidden');

        document.getElementById('hud-quest-tracker').classList.remove('hidden');

        score = 0;

        timeRemaining = GAME_DURATION;

        items = [];

        particles = [];

        sessionMetrics = { garrofoCount: 0, chorizoCaught: 0 };

        document.getElementById('hud-score').innerText = '0';

        document.getElementById('hud-timer').innerText = GAME_DURATION;

        isPlaying = true;

        pan.x = canvas.width / 2;

        pan.targetX = canvas.width / 2;

        if (gameTimer) clearInterval(gameTimer);

        gameTimer = setInterval(() => {

          timeRemaining--;

          document.getElementById('hud-timer').innerText = timeRemaining;

          if (timeRemaining <= 0) endGameSession();

        }, 1000);

        requestAnimationFrame(gameLoop);

      });

    }

    function spawnItem() {

      let rand = Math.random();

      let selectedKey = "RICE";

      if (rand < 0.12) selectedKey = "CHORIZO";

      else if (rand < 0.28) selectedKey = "KETCHUP";

      else if (rand < 0.40) selectedKey = "NOODLES";

      else if (rand < 0.55) selectedKey = "GARROFO";

      else if (rand < 0.70) selectedKey = "GREEN_BEAN";

      else if (rand < 0.85) selectedKey = "CHICKEN";

      else selectedKey = "RABBIT";

      const typeData = INGREDIENT_TYPES[selectedKey];

      items.push({

        x: Math.random() * (canvas.width - 60) + 30,

        y: -30,

        speed: Math.random() * 2.5 + 3.5,

        key: selectedKey,

        ...typeData

      });

    }

    function createParticles(x, y, color) {

      for (let i = 0; i < 10; i++) {

        particles.push({

          x: x, y: y,

          vx: (Math.random() - 0.5) * 6,

          vy: (Math.random() - 0.5) * 6,

          radius: Math.random() * 4 + 2,

          color: color, alpha: 1

        });

      }

    }

    let spawnCounter = 0;

    function gameLoop() {

      if (!isPlaying) return;

      ctx.clearRect(0, 0, canvas.width, canvas.height);

      pan.x += (pan.targetX - pan.x) * 0.25;

      if (pan.x < pan.width / 2) pan.x = pan.width / 2;

      if (pan.x > canvas.width - pan.width / 2) pan.x = canvas.width - pan.width / 2;

      ctx.beginPath();

      ctx.ellipse(pan.x, pan.y + 20, pan.width / 2, 12, 0, 0, Math.PI * 2);

      ctx.fillStyle = 'rgba(0,0,0,0.18)';

      ctx.fill();

      ctx.fillStyle = '#263238';

      ctx.fillRect(pan.x - pan.width / 2 - 15, pan.y - 4, 15, 8);

      ctx.fillRect(pan.x + pan.width / 2, pan.y - 4, 15, 8);

      ctx.beginPath();

      ctx.ellipse(pan.x, pan.y, pan.width / 2, pan.height / 2, 0, 0, Math.PI * 2);

      ctx.fillStyle = '#ffb300';

      ctx.fill();

      ctx.lineWidth = 4;

      ctx.strokeStyle = '#e65100';

      ctx.stroke();

      ctx.beginPath();

      ctx.ellipse(pan.x, pan.y + 4, pan.width / 2 - 8, pan.height / 2 - 6, 0, 0, Math.PI * 2);

      ctx.fillStyle = '#ffe082';

      ctx.fill();

      spawnCounter++;

      if (spawnCounter % 32 === 0) spawnItem();

      for (let i = items.length - 1; i >= 0; i--) {

        let item = items[i];

        item.y += item.speed;

        ctx.beginPath();

        ctx.arc(item.x, item.y, item.radius, 0, Math.PI * 2);

        ctx.fillStyle = item.color;

        ctx.fill();

        ctx.lineWidth = 2;

        ctx.strokeStyle = '#333';

        ctx.stroke();

        const inX = item.x > pan.x - pan.width / 2 && item.x < pan.x + pan.width / 2;

        const inY = item.y + item.radius >= pan.y - pan.height / 4 && item.y - item.radius <= pan.y + pan.height / 2;

        if (inX && inY) {

          if (item.type === 'good') {

            score += item.points;

            audio.playCatchGood();

            createParticles(item.x, item.y, item.color);

            if (item.key === 'GARROFO') sessionMetrics.garrofoCount++;

          } else if (item.type === 'bad') {

            score = Math.max(0, score + item.points);

            audio.playCatchBad();

            createParticles(item.x, item.y, '#d62828');

          } else if (item.type === 'chorizo') {

            score = Math.floor(score * 0.5);

            sessionMetrics.chorizoCaught++;

            audio.playChorizoPenalty();

            createParticles(item.x, item.y, '#800000');

          }

          document.getElementById('hud-score').innerText = score;

          items.splice(i, 1);

          continue;

        }

        if (item.y > canvas.height + 40) items.splice(i, 1);

      }

      for (let i = particles.length - 1; i >= 0; i--) {

        let p = particles[i];

        p.x += p.vx;

        p.y += p.vy;

        p.alpha -= 0.03;

        if (p.alpha <= 0) { particles.splice(i, 1); continue; }

        ctx.save();

        ctx.globalAlpha = p.alpha;

        ctx.beginPath();

        ctx.arc(p.x, p.y, p.radius, 0, Math.PI * 2);

        ctx.fillStyle = p.color;

        ctx.fill();

        ctx.restore();

      }

      requestAnimationFrame(gameLoop);

    }

    function endGameSession() {

      isPlaying = false;

      clearInterval(gameTimer);

      document.getElementById('hud').classList.add('hidden');

      document.getElementById('hud-quest-tracker').classList.add('hidden');

      let questBonusTotal = 0;

      userData.quests.forEach(q => {

        if (!q.completed) {

          if (q.key === 'GARROFO') {

            q.progress += sessionMetrics.garrofoCount;

            if (q.progress >= q.target) { q.progress = q.target; q.completed = true; questBonusTotal += q.reward; }

          } else if (q.key === 'SCORE') {

            if (score >= q.target) { q.progress = q.target; q.completed = true; questBonusTotal += q.reward; }

            else { q.progress = Math.max(q.progress, score); }

          } else if (q.key === 'NO_CHORIZO') {

            if (sessionMetrics.chorizoCaught === 0) { q.progress = 1; q.completed = true; questBonusTotal += q.reward; }

          }

        }

      });

      const sessionFinalPoints = score + questBonusTotal;

      userData.totalPoints += sessionFinalPoints;

      userData.sessionsPlayed++;

      if (score > userData.bestRun) userData.bestRun = score;

      saveUserData();

      document.getElementById('res-score').innerText = score;

      document.getElementById('res-quests').innerText = `+${questBonusTotal}`;

      document.getElementById('res-rank').innerText = getRank(userData.totalPoints);

      document.getElementById('screen-results').classList.remove('hidden');

    }
